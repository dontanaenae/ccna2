config t
vlan 21
 name FINANCE
 exit
Int vlan 21
 no shut
 ip add 10.91.0.1 255.255.255.128
ip ospf 1 area 0
ip dhcp excluded-add 10.91.0.1 10.91.0.100
ip dhcp pool FINANCE.com
 network 10.91.0.0 255.255.255.128
 default-router 10.91.0.1
 domain-name FINANCE.com
 dns-server 10.91.1.10
 option 150 ip 10.91.100.8
 Int Fa 0/7
  swi Voice vlan 21
  do sh ip dhcp binding
